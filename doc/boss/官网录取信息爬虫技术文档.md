## 1. 概述

### 1.1 项目目标

开发一个通用的高校招生信息爬虫系统，能够自动从各高校官网爬取历年录取分数线数据，并将数据标准化后导出为CSV格式。

### 1.2 核心功能

- ✅ 支持多种数据源格式（API接口、HTML页面）
- ✅ 自动识别和解析不同年份的数据格式
- ✅ 数据标准化和清洗
- ✅ 自动添加院校标签（985/211/双一流）
- ✅ CSV格式导出
- ✅ 定时任务支持

### 1.3 支持的学校类型

1. **API接口型**：通过AJAX接口获取JSON数据
   - 示例：北京理工大学、北京航空航天大学

2. **HTML页面型**：通过解析HTML页面获取数据
   - 表格格式：北京大学
   - 段落格式：清华大学

```python
def clean_data(self, data: List[Dict]) -> List[Dict]:
    """数据清洗主流程"""
    cleaned_data = []
    
    for item in data:

        # 2. 清洗分数数据

        # 4. 验证数据
        if self.validate_data(item):
```

```python
class ExampleAPICrawler:
    """API接口型爬虫示例"""
    
    def __init__(self):
        self.base_url = 'https://example.edu.cn'
        self.headers = {
            'User-Agent': 'Mozilla/5.0 ...',
            'Content-Type': 'application/x-www-form-urlencoded; charset=UTF-8',
            'Referer': 'https://example.edu.cn/admission',
            'X-Requested-With': 'XMLHttpRequest'
        }
        self.session = requests.Session()
        self.session.headers.update(self.headers)
        self.school_name = '示例大学'
        self.school_code = '10000'
    
    def get_filter_params(self) -> Optional[Dict]:
        """获取筛选参数"""
        try:
            url = f'{self.base_url}/api/filter'
            response = self.session.post(url, timeout=30)
            response.raise_for_status()
            data = response.json()
            if data.get('success'):
                return data.get('data', {})
            return None
        except Exception as e:
            logger.error(f"获取筛选参数失败: {e}")
            return None
    
    def get_admission_data(self, params: Dict) -> Optional[Dict]:
        """获取招生数据"""
        try:
            url = f'{self.base_url}/api/admission'
            response = self.session.post(url, data=params, timeout=30)
            response.raise_for_status()
            data = response.json()
            if data.get('success'):
                return data.get('data', {})
            return None
        except Exception as e:
            logger.error(f"获取招生数据失败: {e}")
            return None
    
    def parse_batch_data(self, batch_list: List[Dict], year: int, province: str) -> List[Dict]:
        """解析普通批数据"""
        admission_list = []
        
        def get_value_or_na(value, default=''):
            if value is None:
                return default if default else 'NA'
            value_str = str(value).strip()
            return value_str if value_str else (default if default else 'NA')
        
        for item in batch_list:
            min_score = item.get('minScore', '')
            min_rank = item.get('minRank', '')
            
            # 处理分数
            if isinstance(min_score, (int, float)):
                min_score = str(int(min_score))
            elif min_score:
                min_score = str(min_score)
            else:
                min_score = 'NA'
            
            # 处理排名
            if isinstance(min_rank, (int, float)):
                min_rank = str(int(min_rank))
            elif min_rank:
                min_rank = str(min_rank)
            else:
                min_rank = 'NA'
            
            admission_info = {
                '年份': str(item.get('year', year)),
                '学校': self.school_name,
                '_985': '1',
                '_211': '1',
                '双一流': '1',
                '科类': get_value_or_na(item.get('category')),
                '批次': '普通批',
                '专业': 'NA',
                '最低分': min_score,
                '最低分排名': min_rank,
                '全国统一招生代码': self.school_code,
                '招生类型': get_value_or_na(item.get('type'), '统招'),
                '生源地': get_value_or_na(item.get('province', province))
            }
            admission_list.append(admission_info)
        
        return admission_list
    
    def crawl_by_year(self, year: int) -> List[Dict]:
        """爬取指定年份的数据"""
        all_data = []
        
        # 获取筛选参数
        filter_params = self.get_filter_params()
        if not filter_params:
            logger.warning("无法获取筛选参数")
            return []
        
        # 获取省份列表
        provinces = filter_params.get('provinces', [])
        
        # 遍历省份
        for province in provinces:
            params = {
                'year': str(year),
                'province': province
            }
            
            data = self.get_admission_data(params)
            if data:
                batch_list = data.get('batchList', [])
                if batch_list:
                    batch_data = self.parse_batch_data(batch_list, year, province)
                    all_data.extend(batch_data)
            
            time.sleep(0.5)  # 避免请求过快
        
        logger.info(f"共爬取 {year}年 {len(all_data)} 条招生信息")
        return all_data
```

```python
class ExampleTableCrawler:
    """HTML表格型爬虫示例"""
    
    def __init__(self):
        self.base_url = 'https://example.edu.cn'
        self.headers = {
            'User-Agent': 'Mozilla/5.0 ...',
        }
        self.session = requests.Session()
        self.session.headers.update(self.headers)
        self.school_name = '示例大学'
        self.school_code = '10000'
    
    def get_year_url(self, year: int) -> str:
        """获取指定年份的URL"""
        return f'{self.base_url}/admitline/{year}.html'
    
    def parse_score_page(self, url: str, year: int) -> List[Dict]:
        """解析表格页面"""
        try:
            response = self.session.get(url, timeout=30)
            response.encoding = 'utf-8'
            response.raise_for_status()
            
            soup = BeautifulSoup(response.text, 'html.parser')
            admission_list = []
            
            # 查找表格
            table = soup.find('table')
            if not table:
                logger.warning("未找到表格")
                return []
            
            # 解析表头
            headers = []
            header_row = table.find('tr')
            if header_row:
                header_cells = header_row.find_all(['th', 'td'])
                headers = [cell.get_text(strip=True) for cell in header_cells]
            
            # 解析数据行
            rows = table.find_all('tr')[1:]
            for row in rows:
                cells = row.find_all(['td', 'th'])
                if len(cells) < 2:
                    continue
                
                cell_texts = [cell.get_text(strip=True) for cell in cells]
                
                # 根据表头映射数据
                province = ''
                category = ''
                arts_score = ''
                science_score = ''
                other_score = ''
                
                if headers:
                    for idx, header in enumerate(headers):
                        if idx < len(cell_texts):
                            value = cell_texts[idx]
                            if '省份' in header:
                                province = value
                            elif '类别' in header:
                                category = value
                            elif '文科' in header:
                                arts_score = value
                            elif '理科' in header:
                                science_score = value
                            elif '其它' in header or '其他' in header:
                                other_score = value
                
                # 创建记录
                if arts_score and arts_score != '-':
                    score_match = re.search(r'(\d+)', arts_score)
                    if score_match:
                        admission_info = self._create_admission_info(
                            year, province, category, score_match.group(1), '本科一批', '文史'
                        )
                        if admission_info:
                            admission_list.append(admission_info)
                
                # 类似处理理科分数和其他分数...
            
            return admission_list
            
        except Exception as e:
            logger.error(f"解析页面失败: {e}")
            return []
    
    def _create_admission_info(self, year: int, province: str, category: str,
                               score: str, batch: str, category_type: str) -> Optional[Dict]:
        """创建标准格式的招生信息"""
        if not province or not score:
            return None
        
        return {
            '年份': str(year),
            '学校': self.school_name,
            '_985': '1',
            '_211': '1',
            '双一流': '1',
            '科类': category_type,
            '批次': batch,
            '专业': category if category and category != '-' else 'NA',
            '最低分': score,
            '最低分排名': 'NA',
            '全国统一招生代码': self.school_code,
            '招生类型': '统招',
            '生源地': province.replace('省', '').replace('市', '')
        }
    
    def crawl_by_year(self, year: int) -> List[Dict]:
        """爬取指定年份的数据"""
        url = self.get_year_url(year)
        return self.parse_score_page(url, year)
```

### 10.1 请求相关问题

**Q1: 返回403 Forbidden错误**

**解决方案**：
- 检查请求头是否完整（User-Agent、Referer等）
- 尝试先访问主页面建立session
- 检查是否需要特定的Cookie

```python
# 先访问主页面
self.session.get(self.base_url)
# 再访问数据页面
response = self.session.get(data_url)
```

**Q2: 请求超时**

**解决方案**：
- 增加timeout时间
- 添加重试机制

```python
def get_with_retry(self, url, max_retries=3):
    for i in range(max_retries):
        try:
            response = self.session.get(url, timeout=30)
            return response
        except requests.exceptions.Timeout:
            if i == max_retries - 1:
                raise
            time.sleep(2 ** i)  # 指数退避
```

### 10.2 数据解析问题

**Q3: 无法找到表格或数据**

**解决方案**：
- 检查HTML结构是否变化
- 尝试多种选择器
- 保存HTML文件用于调试

```python
# 尝试多种方式查找
```

### 10.3 数据质量问题

**Q5: 数据缺失或不完整**

**解决方案**：
- 使用`NA`填充缺失字段
- 添加数据验证
- 记录警告日志

```python
def validate_data(self, item: Dict) -> bool:
    """验证数据完整性"""
    required_fields = ['年份', '学校', '生源地', '最低分']
    for field in required_fields:
```

**Q6: 分数格式不一致**

**解决方案**：

- 提取数字部分
- 处理小数分数

```python
def clean_score(self, score_str: str) -> str:
    """清洗分数数据"""
    if not score_str or score_str == '-':
        return 'NA'
    
    # 提取数字（处理小数）
    score_match = re.search(r'(\d+)\.?\d*', str(score_str))
    if score_match:
        return str(int(float(score_match.group(1))))
    
    return 'NA'
```
